local Pages = {}

local function createPage(name)
	local page = Instance.new("ScrollingFrame")
	page.Name = name .. "_Page"
	page.Size = UDim2.new(1, 0, 1, 0)
	page.BackgroundTransparency = 1
	page.ScrollBarThickness = 4
	page.CanvasSize = UDim2.new(0, 0, 0, 0)
	page.AutomaticCanvasSize = Enum.AutomaticSize.Y
	page.Visible = false
	page.ZIndex = 5
	page.Parent = ContentArea

	local layout = Instance.new("UIListLayout")
	layout.Padding = UDim.new(0, 6)
	layout.Parent = page

	local pad = Instance.new("UIPadding")
	pad.PaddingTop = UDim.new(0, 8)
	pad.PaddingLeft = UDim.new(0, 8)
	pad.PaddingRight = UDim.new(0, 8)
	pad.PaddingBottom = UDim.new(0, 10)
	pad.Parent = page

	Pages[name] = page
	return page
end

local function switchTab(name)
	for tabName, page in pairs(Pages) do
		page.Visible = (tabName == name)
	end
	for tabName, btn in pairs(TabButtons) do
		if tabName == name then
			btn.BackgroundColor3 = Color3.fromRGB(26, 128, 72)
			btn.TextColor3 = Color3.fromRGB(255, 250, 230)
		else
			btn.BackgroundColor3 = Color3.fromRGB(36, 26, 34)
			btn.TextColor3 = Color3.fromRGB(215, 210, 220)
		end
	end
end

local function addTab(name)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 0, 32)
	btn.BackgroundColor3 = Color3.fromRGB(36, 26, 34)
	btn.Text = name
	btn.TextColor3 = Color3.fromRGB(215, 210, 220)
	btn.Font = Enum.Font.GothamSemibold
	btn.TextSize = 11
	btn.ZIndex = 5
	btn.Parent = Sidebar
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
	TabButtons[name] = btn

	local page = createPage(name)
	btn.MouseButton1Click:Connect(function()
		switchTab(name)
	end)
	return page
end

local function addToggle(parent, text, default, callback)
	local active = default
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 0, 32)
	btn.BackgroundColor3 = active and Color3.fromRGB(28, 132, 74) or Color3.fromRGB(34, 30, 40)
	btn.Text = (active and "🎄 [ON]  " or "❄️ [OFF] ") .. text
	btn.TextColor3 = Color3.fromRGB(245, 245, 250)
	btn.Font = Enum.Font.GothamSemibold
	btn.TextSize = 11
	btn.ZIndex = 6
	btn.Parent = parent
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)

	btn.MouseButton1Click:Connect(function()
		active = not active
		btn.BackgroundColor3 = active and Color3.fromRGB(28, 132, 74) or Color3.fromRGB(34, 30, 40)
		btn.Text = (active and "🎄 [ON]  " or "❄️ [OFF] ") .. text
		callback(active)
	end)
	return btn
end

local function buyPet(name)
	for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
		local n = string.lower(obj.Name)
		if string.find(n, "pet") and (string.find(n, "buy") or string.find(n, "purchase") or string.find(n, "shop")) then
			if obj:IsA("RemoteEvent") then
				pcall(function() obj:FireServer(name) end)
			elseif obj:IsA("RemoteFunction") then
				pcall(function() obj:InvokeServer(name) end)
			end
		end
	end
end

task.spawn(function()
	while task.wait(State.RepInterval) do
		if State.AutoRep1500 then
			local char = player.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")
			if State.UseWeights and char then
				local tool = char:FindFirstChildOfClass("Tool")
				if not tool and player:FindFirstChild("Backpack") then
					for _, item in ipairs(player.Backpack:GetChildren()) do
						if item:IsA("Tool") then
							item.Parent = char
							tool = item
							break
						end
					end
				end
				if tool then
					for _ = 1, State.BurstPerTick do
						pcall(function() tool:Activate() end)
					end
				end
			end
			if State.UseMachines and root then
				for _, obj in ipairs(Workspace:GetDescendants()) do
					if obj:IsA("ProximityPrompt") then
						local pName = string.lower(obj.Name .. " " .. obj.ActionText .. " " .. (obj.Parent and obj.Parent.Name or ""))
						if string.find(pName, "bench") or string.find(pName, "machine") or string.find(pName, "lift") or string.find(pName, "train") then
							local part = obj.Parent:IsA("BasePart") and obj.Parent or obj.Parent:FindFirstChildWhichIsA("BasePart")
							if part and (root.Position - part.Position).Magnitude <= 35 then
								pcall(function()
									if fireproximityprompt then fireproximityprompt(obj) end
								end)
							end
						end
					end
				end
			end
			for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
				if obj:IsA("RemoteEvent") then
					local n = string.lower(obj.Name)
					if string.find(n, "rep") or string.find(n, "lift") or string.find(n, "muscle") or string.find(n, "machine") then
						for _ = 1, math.min(State.BurstPerTick, 25) do
							pcall(function() obj:FireServer("rep") end)
						end
					end
				end
			end
		end
	end
end)

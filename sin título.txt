local function makeToggle(key, label)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 0, 29)
	btn.BackgroundColor3 = State[key] and Color3.fromRGB(26, 128, 72) or Color3.fromRGB(36, 32, 46)
	btn.Text = (State[key] and "[ON]  " or "[OFF] ") .. label
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.Font = Enum.Font.GothamSemibold
	btn.TextSize = 11
	btn.Parent = listFrame
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 7)

	btn.MouseButton1Click:Connect(function()
		State[key] = not State[key]
		btn.BackgroundColor3 = State[key] and Color3.fromRGB(26, 128, 72) or Color3.fromRGB(36, 32, 46)
		btn.Text = (State[key] and "[ON]  " or "[OFF] ") .. label
	end)
end

local function getTool(keyword)
	local char = player.Character
	if not char then return nil end
	local equipped = char:FindFirstChildOfClass("Tool")
	if equipped and (not keyword or string.find(string.lower(equipped.Name), keyword)) then
		return equipped
	end
	if State.AutoEquipTool and player:FindFirstChild("Backpack") then
		for _, item in ipairs(player.Backpack:GetChildren()) do
			if item:IsA("Tool") and (not keyword or string.find(string.lower(item.Name), keyword)) then
				item.Parent = char
				return item
			end
		end
	end
	return equipped
end

-- 1. Fast Rep 1500 cada 0.1s (150 por tick * 10 ticks/seg = 1500/s)
task.spawn(function()
	while task.wait(0.1) do
		if State.FastRep1500 then
			pcall(function()
				local char = player.Character
				local root = char and char:FindFirstChild("HumanoidRootPart")
				local tool = getTool(nil)
				if tool then
					for _ = 1, 150 do
						tool:Activate()
					end
				end
				if root then
					for _, obj in ipairs(Workspace:GetDescendants()) do
						if obj:IsA("ProximityPrompt") then
							local pName = string.lower(obj.Name .. " " .. obj.ActionText .. " " .. (obj.Parent and obj.Parent.Name or ""))
							if string.find(pName, "bench") or string.find(pName, "machine") or string.find(pName, "lift") or string.find(pName, "train") then
								local part = obj.Parent:IsA("BasePart") and obj.Parent or obj.Parent:FindFirstChildWhichIsA("BasePart")
								if part and (root.Position - part.Position).Magnitude <= 35 and fireproximityprompt then
									fireproximityprompt(obj)
								end
							end
						end
					end
				end
				for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
					if obj:IsA("RemoteEvent") then
						local n = string.lower(obj.Name)
						if string.find(n, "rep") or string.find(n, "lift") or string.find(n, "muscle") or string.find(n, "machine") then
							for _ = 1, 25 do
								obj:FireServer("rep")
							end
						end
					end
				end
			end)
		end
	end
end)

-- 3 y 4. Auto Kill Boss Alien y Auto Kill Bosses + 6. Auto Punch
task.spawn(function()
	while task.wait(0.15) do
		pcall(function()
			local char = player.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")

			if (State.AutoKillAlien or State.AutoKillBosses) and root then
				for _, obj in ipairs(Workspace:GetDescendants()) do
					if obj:IsA("Model") and obj ~= char then
						local hum = obj:FindFirstChildOfClass("Humanoid")
						local bRoot = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildWhichIsA("BasePart")
						if hum and hum.Health > 0 and bRoot then
							local lower = string.lower(obj.Name)
							local isAlien = string.find(lower, "alien") or string.find(lower, "ufo")
							local isBoss = string.find(lower, "boss")
							if (State.AutoKillAlien and isAlien) or (State.AutoKillBosses and (isBoss or isAlien)) then
								root.CFrame = bRoot.CFrame * CFrame.new(0, 0, 4)
								local punch = getTool("punch") or getTool(nil)
								if punch then
									punch:Activate()
								end
								break
							end
						end
					end
				end
			end

			if State.AutoPunch then
				local punch = getTool("punch") or getTool(nil)
				if punch then
					punch:Activate()
				end
			end
		end)
	end
end)

-- 2. Auto Rebirth + 7. Orbs/Gems + 8. Chests
task.spawn(function()
	while task.wait(1) do
		pcall(function()
			if State.AutoRebirth then
				for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
					if obj:IsA("RemoteEvent") and string.find(string.lower(obj.Name), "rebirth") then
						obj:FireServer()
					end
				end
			end

			if State.AutoChests then
				for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
					if obj:IsA("RemoteEvent") then
						local n = string.lower(obj.Name)
						if string.find(n, "chest") or string.find(n, "reward") then
							obj:FireServer()
						end
					end
				end
			end

			if State.AutoCollectOrbs then
				local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
				if root then
					for _, obj in ipairs(Workspace:GetDescendants()) do
						if obj:IsA("BasePart") then
							local n = string.lower(obj.Name)
							if (string.find(n, "orb") or string.find(n, "gem")) and (root.Position - obj.Position).Magnitude <= 120 then
								obj.CFrame = root.CFrame
							end
						end
					end
				end
			end
		end)
	end
end)

-- 9. Fly
RunService.RenderStepped:Connect(function()
	pcall(function()
		local char = player.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if State.Fly and root and hum then
			hum.PlatformStand = true
			local dir = hum.MoveDirection
			local vy = 0
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
				vy = 75
			elseif UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
				vy = -75
			end
			root.AssemblyLinearVelocity = Vector3.new(dir.X * 75, vy, dir.Z * 75)
		elseif hum and hum.PlatformStand and not State.Fly then
			hum.PlatformStand = false
		end
	end)
end)
